async function process() { const input = document.getElementById('urlInput').value; let eatToken = input.includes("?eat=") ? new URLSearchParams(new URL(input).search).get('eat') : input; if(!eatToken) return alert("Sahi URL dalo!"); // Button status change document.querySelector('.btn-main').innerText = "Processing..."; try { // Apne Vercel API ko call karein const res = await fetch(`/api/get-token?eat=${eatToken}`); const data = await res.json(); if(data.access_token) { document.getElementById('tokenTxt').innerText = data.access_token; document.getElementById('resSection').style.display = 'block'; } else { alert("Token generate nahi hua!"); } } catch (e) { alert("API Error!"); } finally { document.querySelector('.btn-main').innerText = "GENERATE ACCESS →"; } }
